{# ABOUTME: Design-system rendering of the optional signup survey fields (SignupSurveyFormMixin) ABOUTME: Switches on the WTForms field type, so it tracks SIGNUP_SURVEY_QUESTIONS without naming any question #} {% from "backoffice/components/input.html" import input, textarea, select, radio_group, first_error %} {% macro signup_survey_fields(form) %} {% for field in form.survey_fields() %} {% if field.type == "RadioField" %} {% set options = [] %} {% for value, label in field.choices %} {% set _ = options.append({"value": value, "label": label}) %} {% endfor %} {{ radio_group(field.name, label=field.label.text, options=options, value=field.data or "", hint=field.description or "", error=first_error(field.errors)) }} {% elif field.type == "SelectField" %} {% set options = [] %} {% for value, label in field.choices %} {% set _ = options.append({"value": value, "label": label}) %} {% endfor %} {{ select(field.name, label=field.label.text, options=options, value=field.data or "", hint=field.description or "", error=first_error(field.errors)) }} {% elif field.type == "TextAreaField" %} {{ textarea(field.name, label=field.label.text, value=field.data or "", hint=field.description or "", error=first_error(field.errors)) }} {% else %} {{ input(field.name, label=field.label.text, value=field.data or "", hint=field.description or "", error=first_error(field.errors)) }} {% endif %} {% endfor %} {% endmacro %}
